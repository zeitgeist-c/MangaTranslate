**E1.2 — `normalizePageBoxes(rawBoxes, imgW, imgH)`:** Model koordinat uzayı güvenilmez (v3.2.2 dersi: 0-1 / piksel / binde-birim). Sayfa bazında tek karar ver:
```js
function detectBoxSpace(rawBoxes, imgW, imgH) {
  var max = 0;
  rawBoxes.forEach(function (b) {
    ['x', 'y', 'width', 'height'].forEach(function (k) {
      var n = Number(b && b[k]);
      if (isFinite(n) && n > max) max = n;
    });
  });
  if (max <= 1.05) return 'unit';                       // 0-1
  if (max <= Math.max(imgW, imgH) * 1.05) return 'pixel';
  return 'unit1000';                                    // binde-birim
}
function normalizePageBoxes(rawBoxes, imgW, imgH) {
  var space = detectBoxSpace(rawBoxes, imgW, imgH);
  return rawBoxes.map(function (b) {
    var x = Number(b.x), y = Number(b.y), w = Number(b.width), h = Number(b.height);
    if (!isFinite(x) || !isFinite(y) || !isFinite(w) || !isFinite(h)) return null;
    if (space === 'pixel') { x /= imgW; y /= imgH; w /= imgW; h /= imgH; }
    else if (space === 'unit1000') { x /= 1000; y /= 1000; w /= 1000; h /= 1000; }
    return { x: x, y: y, width: w, height: h, text: String(b.text || '').trim() };
  }).filter(Boolean);
}
```

function bandBoxToImage01(box, band, imgW, imgH) {
  return {
    x: (band.x + box.x * band.width) / imgW,
    y: (band.y + box.y * band.height) / imgH,
    width: (box.width * band.width) / imgW,
    height: (box.height * band.height) / imgH,
    text: box.text || '',
  };
}

**E1.4 — `filterCandidates(boxes, opts)`** (çöp kutuları at): min kenar görsel oranı (varsayılan `0.008`), alan üst sınırı `0.9`, tamamen görsel dışı olanlar; `clampBox01` (0..1'e kırp).
```js
function clampBox01(b) {
  var x = Math.max(0, Math.min(1, b.x)), y = Math.max(0, Math.min(1, b.y));
  var x2 = Math.max(0, Math.min(1, b.x + b.width)), y2 = Math.max(0, Math.min(1, b.y + b.height));
  return { x: x, y: y, width: x2 - x, height: y2 - y, text: b.text || '' };
}
function filterCandidates(boxes, opts) {
  var minSide = (opts && opts.minSide) || 0.008;
  var maxArea = (opts && opts.maxArea) || 0.9;
  return boxes
    .map(clampBox01)
    .filter(function (b) {
      if (!(b.width > 0) || !(b.height > 0)) return false;
      if (Math.min(b.width, b.height) < minSide) return false;
      if (b.width * b.height > maxArea) return false;
      return true;
    });
}
```

**E1.5 — `dedupeBoxes(boxes, threshold)`** (bant örtüşmesinde aynı balon iki kez gelebilir):
```js
function iou(a, b) {
  var ix = Math.max(0, Math.min(a.x + a.width, b.x + b.width) - Math.max(a.x, b.x));
  var iy = Math.max(0, Math.min(a.y + a.height, b.y + b.height) - Math.max(a.y, b.y));
  var inter = ix * iy;
  var uni = a.width * a.height + b.width * b.height - inter;
  return uni > 0 ? inter / uni : 0;
}
function dedupeBoxes(boxes, threshold) {
  var t = threshold || 0.45;
  var out = [];
  boxes.forEach(function (b) {
    for (var i = 0; i < out.length; i++) {
      if (iou(out[i], b) >= t) {
        if (!out[i].text && b.text) out[i].text = b.text;   // metni dolu olan kazanir
        return;
      }
    }
    out.push({ x: b.x, y: b.y, width: b.width, height: b.height, text: b.text || '' });
  });
  return out;
}
```

**E1.6 — `parseTranslateResponse(raw, count)`** (toplu çeviri yanıtı → sıralı dizi):
```js
function parseTranslateResponse(raw, count) {
  var data;
  try { data = JSON.parse(String(raw)); } catch (e) { throw new Error('ceviri JSON cozulemedi'); }
  var list = data && data.translations;
  if (!Array.isArray(list)) throw new Error('translations dizisi yok');
  var out = new Array(count).fill(null);
  list.forEach(function (item) {
    var i = Number(item && item.i);
    if (!Number.isInteger(i) || i < 0 || i >= count) return;         // aralik disi sessizce atlanir
    var tr = String((item && (item.tr != null ? item.tr : item.translation)) || '').trim();
    if (tr) out[i] = tr.slice(0, 200);                               // 200 siniri mevcut kuralla ayni
  });
  return out;
}
```

  const apiKey = await getActiveApiKey();

}

function parseDetectArray(raw) {
  if (!raw || !String(raw).trim()) throw new Error('parseDetectArray: bos giris');
  const data = JSON.parse(String(raw));
  const list = (data && data.bubbles) || [];
  if (!Array.isArray(list)) throw new Error('bubbles dizisi yok');
  return list.filter(b => b && typeof b === 'object');
}

## Risks / tradeoffs / open questions

1. **Maket onayı bekliyor**: opaklık `%8/%16/%28` — `DETECT_FILL` tek satır (E3.2). Onay gelmeden E4'ü bitirme.
2. **Görsel alma başarısızlığı**: bazı CDN'ler SW `fetch`'ini de (referer/cookie kontrolü) reddedebilir → görünür alan yedeği (kısmi tarama) devrede; kullanıcıya mesaj gösterilir. Gerçek testte ilk doğrulanacak şey (E7.3).
3. **Koordinat uzayı heuristiği** (`normalizePageBoxes`) binde-birim/piksel ayrımını yanlış yaparsa kutular sistematik kayar; akış gereği kullanıcı düzeltir ama "hiç tutmadı" hissi doğar → E7.2'de 3 sayfada doğrulanır; kayma görülürse eşik (`1.05`) ve uzay sırası core'da ayarlanır (testli).
4. **Toplu çeviri kalitesi** (metin-only, bağlamsız): E7.1'de ölçülür. Yetersizse toplu istekte görüntü de gönderilir: `images:[dataUrl]` + metin listesi → yine 1 istek (E2.3'te `imageBase64` dolu geçirilir). Karar ölçüme bağlı.
5. **Bant başına tespit = bant sayısı istek**: 6 bantlı webtoonda tarama ~30-40 sn; kabul (kullanıcı onaylı akış). Bant sayısı `PM_MAX_BANDS=6` tavanıyla sınırlı (E3.4).
6. **Pencere yeniden boyutlanınca kutu ölçeklenmez** (mevcut Alan Seç davranışıyla aynı) — v1 kabulü; şikayet gelirse yüzde tabanlı yeniden ölçekleme ayrı iş.
7. **Sürüm no `v3.5.0`** (3.2-3.4 arşiv sayfa-modu hattında; çakışma yok) — onay bekliyor.
8. **`callGemini` imageBase64 null**: E2.3'te doğrula; gerekirse tek satır guard.
9. **Önbellek ve anahtar rotasyonu** kullanıcı kararına bağlı (maket butonları); önbellek E8 (koşullu), rotasyon kapsam dışı bırakıldı — istek/sayfa 2'ye indiği için aciliyeti düştü; ölçüm 429 gösterirse ayrı tur.
10. **`.hermes/plans/` dışına dokunulmaz**; araç scriptleri `%TEMP%`'te kalır; repoya geçici dosya bırakılmaz.
